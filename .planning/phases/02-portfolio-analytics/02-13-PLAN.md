```typescript
export type Period = '1M' | '3M' | '6M' | '1Y' | '3Y' | 'all'

export function getPeriodBounds(period: string): { start: Date; end: Date } | null {
  if (period === 'all') return null
  const msMap: Record<string, number> = { '1M': 30, '3M': 90, '6M': 180, '1Y': 365, '3Y': 1095 }
  const days = msMap[period]
  if (!days) return null    // ← 'FY' hits this branch and returns null today — BUG
  const now = new Date()
  return { start: new Date(now.getTime() - days * 86400_000), end: now }
}

export function getCurrentFY(): { start: Date; end: Date } {
  const now = new Date()
  const year = now.getMonth() >= 3 ? now.getFullYear() : now.getFullYear() - 1
  return {
    start: new Date(year, 3, 1),       // April 1 of FY start year
    end:   new Date(year + 1, 2, 31),  // March 31 of next calendar year
  }
}
```
